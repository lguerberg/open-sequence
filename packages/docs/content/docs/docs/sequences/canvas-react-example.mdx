This page is a working sequence editor in about 200 lines of React and a 25-line proxy. Read [Building a canvas](/docs/sequences/building-a-canvas) first: it explains each call this code makes.

It draws nodes as a list of cards in path order, with a select per port to wire it. A real canvas swaps the list for a graph library; the data handling stays the same.

## The proxy on your backend

The browser never holds the API key. Your backend checks its own session, then forwards a short allowlist of calls.

```ts title="server.ts"
import express from 'express';

const app = express();
app.use(express.json({ limit: '300kb' }));

const SEQ = process.env.SEQ_URL!;        // https://sequences.example.com
const KEY = process.env.SEQ_ADMIN_KEY!;  // an admin key: saving and publishing need it
const allowed = [
  /^\/v1\/schema(\/fields)?$/,
  /^\/v1\/sequences:validate$/,
  /^\/v1\/sequences\/[\w-]+\/versions\/[\w-]+(\/draft|\/publish)?$/,
];

app.use('/oseq', async (req, res) => {
  // Check your own user session here before forwarding anything.
  if (!allowed.some((r) => r.test(req.path))) return res.status(404).end();
  const r = await fetch(SEQ + req.path, {
    method: req.method,
    headers: { authorization: `Bearer ${KEY}`, 'content-type': 'application/json' },
    body: req.method === 'GET' ? undefined : JSON.stringify(req.body ?? {}),
  });
  res.status(r.status).type('application/json').send(await r.text());
});
```

## The editor

```tsx title="SequenceEditor.tsx"
import { useEffect, useState } from 'react';

type Problem = { path: string; code: string; message: string; severity: 'error' | 'warning' };
type Node = { type: string; channel?: string; action?: string; params?: Record<string, any>; next: Record<string, string>; branches_by_outcome?: Record<string, string> };
type Def = { entry: string; nodes: Record<string, Node>; [key: string]: unknown };
type Transitions = { keys: string[]; required: string[] };
type Catalogue = {
  nodes: { type: string; label: string; transitions: Transitions; params: any }[];
  channels: { name: string; label: string; actions: { action: string; label: string; transitions: Transitions; params: any }[] }[];
};
type Field = { field: string; label: string; type: string; values?: string[] };

const OPS: Record<string, string[]> = {
  text: ['is', 'is_not', 'contains', 'is_empty', 'is_not_empty'],
  number: ['eq', 'ne', 'gt', 'gte', 'lt', 'lte'],
  bool: ['is_true', 'is_false'],
  enum: ['is', 'is_not'],
  date: ['before', 'after'],
};
const UNARY = ['is_empty', 'is_not_empty', 'is_true', 'is_false'];

async function api<T>(path: string, method = 'GET', body?: unknown): Promise<T> {
  const r = await fetch('/oseq' + path, { method, headers: { 'content-type': 'application/json' }, body: body === undefined ? undefined : JSON.stringify(body) });
  const envelope = await r.json();
  if (!envelope.success) throw Object.assign(new Error(envelope.error.message), envelope.error);
  return envelope.data;
}

const actionOf = (cat: Catalogue, n: Node) => cat.channels.find((c) => c.name === n.channel)?.actions.find((a) => a.action === n.action);
const paramsSchema = (cat: Catalogue, n: Node) => (n.type === 'outreach' ? actionOf(cat, n)?.params : cat.nodes.find((t) => t.type === n.type)?.params);

function ports(cat: Catalogue, n: Node): string[] {
  if (n.type === 'wait_for') return [...(n.params?.events ?? []), ...(n.params?.timeout ? ['timeout'] : [])];
  if (n.type === 'outreach') return actionOf(cat, n)?.transitions.keys ?? [];
  const keys = cat.nodes.find((t) => t.type === n.type)?.transitions.keys ?? [];
  return n.type === 'task' ? [...keys, ...Object.keys(n.branches_by_outcome ?? {})] : keys;
}

/** Path order from the entry; unreachable nodes go last so the user can wire them. */
function order(def: Def): string[] {
  const seen: string[] = [];
  const walk = (id: string) => {
    if (seen.includes(id) || !def.nodes[id]) return;
    seen.push(id);
    const n = def.nodes[id];
    [...Object.values(n.next), ...Object.values(n.branches_by_outcome ?? {})].forEach(walk);
  };
  walk(def.entry);
  return [...seen, ...Object.keys(def.nodes).filter((id) => !seen.includes(id))];
}

export function SequenceEditor({ seqId, versionId }: { seqId: string; versionId: string }) {
  const [cat, setCat] = useState<Catalogue>();
  const [fields, setFields] = useState<Field[]>([]);
  const [def, setDef] = useState<Def>();
  const [lock, setLock] = useState(0);
  const [problems, setProblems] = useState<Problem[]>([]);
  const [status, setStatus] = useState('');
  const base = `/v1/sequences/${seqId}/versions/${versionId}`;

  useEffect(() => {
    Promise.all([api<Catalogue>('/v1/schema'), api<{ sources: { fields: Field[] }[]; attrs: Field[] }>('/v1/schema/fields'), api<any>(base)]).then(([c, f, v]) => {
      setCat(c);
      setFields([...f.sources.flatMap((s) => s.fields), ...f.attrs]);
      setDef(v.version.definition);
      setLock(v.version.lockVersion);
    });
  }, [base]);

  useEffect(() => {
    if (!def) return;
    const t = setTimeout(() => api<{ problems: Problem[] }>('/v1/sequences:validate', 'POST', { definition: def }).then((r) => setProblems(r.problems)), 300);
    return () => clearTimeout(t);
  }, [def]);

  if (!cat || !def) return <p>Loading…</p>;
  const setNode = (id: string, n: Node) => setDef({ ...def, nodes: { ...def.nodes, [id]: n } });
  const addNode = (type: string, channel?: string, action?: string) => {
    let i = 1;
    while (def.nodes[`${channel ?? type}_${i}`]) i++;
    setNode(`${channel ?? type}_${i}`, { type, channel, action, params: {}, next: {} });
  };
  const hasErrors = problems.some((p) => p.severity === 'error');

  async function save(): Promise<boolean> {
    try {
      const r = await api<any>(`${base}/draft`, 'PUT', { definition: def, lock_version: lock });
      setLock(r.version.lockVersion);
      setStatus('Saved');
      return true;
    } catch (e: any) {
      setStatus(e.code === 'STALE_LOCK' ? 'Someone else saved this draft. Reload to see their version.' : e.message);
      return false;
    }
  }
  async function publish() {
    if (!(await save())) return;
    try {
      await api(`${base}/publish`, 'POST', {});
      setStatus('Published');
    } catch (e: any) {
      setStatus(e.message);
    }
  }

  return (
    <div style={{ display: 'grid', gap: 12, maxWidth: 640 }}>
      <div>
        {cat.nodes.filter((t) => t.type !== 'outreach').map((t) => <button key={t.type} onClick={() => addNode(t.type)}>+ {t.label}</button>)}
        {cat.channels.flatMap((c) => c.actions.map((a) => <button key={`${c.name}.${a.action}`} onClick={() => addNode('outreach', c.name, a.action)}>+ {a.label}</button>))}
      </div>
      {problems.filter((p) => !p.path.startsWith('nodes.')).map((p) => <Issue key={p.path + p.code} p={p} />)}
      {order(def).map((id) => {
        const n = def.nodes[id];
        const mine = problems.filter((p) => p.path.split('.')[1] === id && p.path.startsWith('nodes.'));
        const color = mine.some((p) => p.severity === 'error') ? 'crimson' : mine.length ? 'orange' : '#ccc';
        return (
          <section key={id} style={{ border: `2px solid ${color}`, borderRadius: 8, padding: 12 }}>
            <strong>{id}</strong> <small>{n.type}{n.channel ? ` · ${n.channel}` : ''}</small>
            {n.type === 'condition'
              ? <Rules n={n} fields={fields} onChange={(params) => setNode(id, { ...n, params })} />
              : <Params schema={paramsSchema(cat, n)} value={n.params ?? {}} onChange={(params) => setNode(id, { ...n, params })} />}
            {ports(cat, n).map((key) => {
              const isOutcome = n.type === 'task' && !['completed', 'no_answer'].includes(key);
              const target = isOutcome ? n.branches_by_outcome?.[key] : n.next[key];
              const wire = (to: string) => {
                const map = { ...(isOutcome ? n.branches_by_outcome : n.next) };
                if (to) map[key] = to; else delete map[key];
                setNode(id, isOutcome ? { ...n, branches_by_outcome: map } : { ...n, next: map });
              };
              return (
                <div key={key}>
                  {key} →{' '}
                  <select value={target ?? ''} onChange={(e) => wire(e.target.value)}>
                    <option value="">(ends here)</option>
                    {Object.keys(def.nodes).filter((o) => o !== id).map((o) => <option key={o}>{o}</option>)}
                  </select>
                </div>
              );
            })}
            {mine.map((p) => <Issue key={p.path + p.code} p={p} />)}
          </section>
        );
      })}
      <div>
        <button onClick={save}>Save draft</button>{' '}
        <button onClick={publish} disabled={hasErrors}>Publish</button> {status}
      </div>
    </div>
  );
}

function Issue({ p }: { p: Problem }) {
  return <p style={{ color: p.severity === 'error' ? 'crimson' : 'darkorange', margin: '4px 0' }}>{p.message}</p>;
}

/** A form from the JSON Schema of the node's params: strings, booleans and enums. */
function Params({ schema, value, onChange }: { schema: any; value: Record<string, any>; onChange: (v: Record<string, any>) => void }) {
  return (
    <>
      {Object.entries<any>(schema?.properties ?? {}).map(([key, prop]) => {
        const set = (v: unknown) => onChange({ ...value, [key]: v });
        const current = value[key] ?? prop.default ?? '';
        if (prop.enum) return <label key={key}> {key} <select value={current} onChange={(e) => set(e.target.value)}><option value="" />{prop.enum.map((o: string) => <option key={o}>{o}</option>)}</select></label>;
        if (prop.type === 'boolean') return <label key={key}> <input type="checkbox" checked={!!current} onChange={(e) => set(e.target.checked)} /> {key}</label>;
        if (prop.type === 'string') return <label key={key} style={{ display: 'block' }}>{key}<br /><textarea rows={key === 'body' ? 4 : 1} style={{ width: '100%' }} value={current} onChange={(e) => set(e.target.value)} /></label>;
        return null; // arrays (wait_for events) and numbers: add inputs as you need them
      })}
    </>
  );
}

/** The condition field picker. attrs.* and open crm.* paths validate as text. */
function Rules({ n, fields, onChange }: { n: Node; fields: Field[]; onChange: (p: Record<string, any>) => void }) {
  const params = { mode: 'all', rules: [{ field: '', op: 'is', value: '' }], ...n.params };
  const typeOf = (f: string) => (f.startsWith('attrs.') ? 'text' : fields.find((x) => x.field === f)?.type ?? 'text');
  const setRule = (i: number, r: object) => onChange({ ...params, rules: params.rules.map((x: any, j: number) => (j === i ? { ...x, ...r } : x)) });
  return (
    <div>
      <select value={params.mode} onChange={(e) => onChange({ ...params, mode: e.target.value })}><option value="all">all of</option><option value="any">any of</option></select>
      {params.rules.map((r: any, i: number) => (
        <div key={i}>
          <select value={r.field} onChange={(e) => setRule(i, { field: e.target.value, op: OPS[typeOf(e.target.value)][0], value: '' })}>
            <option value="">pick a field</option>
            {fields.map((f) => <option key={f.field} value={f.field}>{f.label}</option>)}
          </select>
          <select value={r.op} onChange={(e) => setRule(i, { op: e.target.value })}>{OPS[typeOf(r.field)].map((o) => <option key={o}>{o}</option>)}</select>
          {!UNARY.includes(r.op) && <input value={r.value ?? ''} onChange={(e) => setRule(i, { value: e.target.value })} />}
        </div>
      ))}
      <button onClick={() => onChange({ ...params, rules: [...params.rules, { field: '', op: 'is', value: '' }] })}>+ rule</button>
    </div>
  );
}
```

## What it leaves out

- **Free paths.** The picker lists only registered fields and your `attrs_schema`. Add a text input for any `attrs.<path>` or `crm.<path>`.
- **Numbers.** Rule values are sent as strings. The validator accepts a numeric string for a number field, and the comparison converts it.
- **Settings.** `settings.send_window`, `jitter`, `reenroll` and `exit_on` need a small form of their own; their problems land in the top list.
- **Task outcomes.** Add an input that creates a `branches_by_outcome` key; it then shows as a port.
- **Concurrent edits.** On `STALE_LOCK` this example only tells the user. Reload the version and offer to reapply the change.
